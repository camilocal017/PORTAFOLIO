// finances.ts — break-even
export function buildFinancialSummary(
  input: FinancialInput
): FinancialSummary {
  const grossProfit =
    input.monthRevenue - input.cogs
  const marginPct =
    grossProfit / input.monthRevenue * 100
  const breakEven =
    input.fixedExpenses / (marginPct / 100)
  return { grossProfit, marginPct,
    breakEvenReached:
      input.monthRevenue >= breakEven }
}
